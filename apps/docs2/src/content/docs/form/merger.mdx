The merger is responsible for combining JSON Schema definitions, merging `allOf` compositions,
and applying schema defaults to form data.

```typescript
export interface Merger {
  mergeSchemas(a: Schema, b: Schema): Schema;
  /**
   * Merges schema and its `allOf` schemas into a single schema
   */
  mergeAllOf(schema: Schema): Schema;
}

export interface MergeFormDataAndSchemaDefaultsOptions {
  schema: Schema;
  formData: SchemaValue | undefined;
  initialDefaultsGenerated?: boolean;
  includeUndefinedValues?: boolean | "excludeObjectChildren";
}

export interface FormValueChange {
  path: FieldPath;
  oldValue: SchemaValue | undefined;
}

export interface DroppedDefault {
  path: RPath;
  value: FieldValue;
  /**
   * The schema that described the path while it held the value, which is what
   * a replacement the defaults fill writes is judged against.
   */
  writtenUnder: Schema | undefined;
}

export interface DropStaleDefaultsResult {
  formData: SchemaValue | undefined;
  /**
   * The values removed, in walk order. A path's parent is never listed
   * alongside it.
   */
  dropped: readonly DroppedDefault[];
}

export interface DropStaleDefaultsOptions {
  validator: Validator;
  rootSchema: Schema;
  /**
   * The schema resolved for the current form data.
   */
  newSchema: Schema;
  formData: SchemaValue | undefined;
  /**
   * Values written since the last defaults pass. A value at one of these paths,
   * or below one, belongs to the user.
   */
  changes: readonly FormValueChange[];
  /**
   * The form data as it arrived, before this defaults pass dropped anything.
   * When omitted, it is rebuilt from `changes`.
   */
  oldFormData?: SchemaValue | undefined;
  /**
   * The schema resolved for `oldFormData`. When omitted, it is retrieved from
   * `rootSchema` and `oldFormData`.
   */
  oldSchema?: Schema | undefined;
  /**
   * Paths this pass has already dropped. They are skipped so a replacement the
   * defaults fill has written is not dropped again.
   */
  droppedPaths?: ReadonlySet<FieldPath>;
  /**
   * Returns the canonical `FieldPath` for `key` below `parentPath`. The form
   * supplies its own so `droppedPaths` can be compared by identity.
   */
  getChildPath?: (parentPath: RPath, key: string | number) => FieldPath;
  /**
   * The flag the defaults fill is running with. The search computes the same
   * defaults as that fill, so it must use the same flag.
   * @default false
   */
  initialDefaultsGenerated?: boolean;
}

export interface FormMerger extends Merger {
  /**
   * Merges defaults of `schema` into `formData`
   */
  mergeFormDataAndSchemaDefaults(
    options: MergeFormDataAndSchemaDefaultsOptions
  ): SchemaValue | undefined;
  /**
   * Removes values that still hold a `default` or `const` declared by a
   * `dependencies` or `if`/`then`/`else` branch that is no longer selected, so
   * the next `mergeFormDataAndSchemaDefaults` call writes the newly selected
   * branch's default. The returned entries let the form put back a value no
   * fill answered and judge a replacement at its own key.
   */
  dropStaleDefaults(options: DropStaleDefaultsOptions): DropStaleDefaultsResult;
}
```

Use `createFormMerger` from `@sjsf/form/mergers/modern` for most projects.
The legacy merger is available for v2 compatibility.

The built-in mergers implement `dropStaleDefaults` with the exported
`dropStaleDefaults(merger, options)` helper. Pass the `FormMerger` itself so the
helper computes defaults exactly as your `mergeFormDataAndSchemaDefaults` does.
A custom `FormMerger` should delegate to the same helper:

```typescript
import {
  dropStaleDefaults,
  type DropStaleDefaultsOptions,
  type FormMerger,
} from "@sjsf/form";

const merger: FormMerger = {
  // ...
  dropStaleDefaults(options: DropStaleDefaultsOptions) {
    return dropStaleDefaults(merger, options);
  },
};
```

The helper is the reference implementation; the interface comments above
describe every option and result field a custom implementation must honor.
